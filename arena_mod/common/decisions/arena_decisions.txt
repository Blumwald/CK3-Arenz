# Arena Decisions

arena_construct_decision = {
	major = yes
	desc = arena_construct_decision_desc
	selection_tooltip = arena_construct_decision_tooltip
	
	is_shown = {
		is_ruler = yes
		is_landed = yes
		exists = capital_county
		capital_county = {
			NOT = { has_county_modifier = arena_location }
		}
	}
	
	is_valid = {
		gold >= 200
		is_at_war = no
	}
	
	effect = {
		remove_short_term_gold = 200
		capital_county = {
			add_county_modifier = {
				modifier = arena_location
				years = 100
			}
		}
		add_prestige = 100
	}
	
	ai_check_interval = 120
	
	ai_potential = {
		is_at_war = no
		gold >= 500
	}
	
	ai_will_do = {
		base = 10
	}
}

arena_enter_decision = {
	desc = arena_enter_decision_desc
	selection_tooltip = arena_enter_decision_tooltip
	
	is_shown = {
		exists = location
		location = {
			county = {
				has_county_modifier = arena_location
			}
		}
		is_adult = yes
		is_imprisoned = no
	}
	
	is_valid = {
		NOT = { has_trait = wounded }
		NOT = { has_trait = maimed }
	}
	
	effect = {
		# Initialize arena stats if first time
		if = {
			limit = { NOT = { has_variable = arena_wins } }
			set_variable = {
				name = arena_wins
				value = 0
			}
			set_variable = {
				name = arena_losses
				value = 0
			}
			add_trait = arena_rank_f
		}
		
		# Create arena opponent
		create_character = {
			location = root.capital_province
			dynasty = none
			age = { 25 35 }
			gender_female_chance = 50
			random_traits = yes
			trait = education_martial_2
			save_scope_as = arena_opponent
		}
		
		# Start duel
		start_duel = {
			target = scope:arena_opponent
			skill = prowess
			value = high_duel_value
		}
		
		# Set flag to track duel result
		set_character_flag = {
			flag = in_arena_duel
			days = 1
		}
	}
	
	ai_check_interval = 60
	
	ai_potential = {
		is_adult = yes
		is_imprisoned = no
		NOT = { has_trait = wounded }
	}
	
	ai_will_do = {
		base = 10
		
		modifier = {
			factor = 2
			has_trait = brave
		}
		
		modifier = {
			factor = 0.5
			has_trait = craven
		}
	}
}
