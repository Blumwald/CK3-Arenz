\xef\xbb\xbf# Arena Decisions

arena_construct_decision = {
	major = yes
	desc = arena_construct_decision_desc
	selection_tooltip = arena_construct_decision_tooltip
	
	is_shown = {
		is_ruler = yes
		is_landed = yes
		exists = capital_county
		capital_county = {
			NOT = { has_county_modifier = arena_location }
		}
	}
	
	is_valid = {
		gold >= 200
		is_at_war = no
	}
	
	effect = {
		remove_short_term_gold = 200
		capital_county = {
			add_county_modifier = {
				modifier = arena_location
				years = 100
			}
		}
		add_prestige = 100
	}
	
	ai_check_interval = 120
	
	ai_potential = {
		is_at_war = no
		gold >= 500
	}
	
	ai_will_do = {
		base = 10
	}
}

arena_enter_decision = {
	desc = arena_enter_decision_desc
	selection_tooltip = arena_enter_decision_tooltip
	
	is_shown = {
		exists = location
		location = {
			county = {
				has_county_modifier = arena_location
			}
		}
		is_adult = yes
		is_imprisoned = no
	}
	
	is_valid = {
		NOT = { has_trait = wounded }
		NOT = { has_trait = maimed }
	}
	
	effect = {
		# Initialize arena stats if first time
		if = {
			limit = { NOT = { has_variable = arena_wins } }
			set_variable = {
				name = arena_wins
				value = 0
			}
			set_variable = {
				name = arena_losses
				value = 0
			}
			add_trait = arena_rank_f
		}
		
		# Simple combat resolution
		random = {
			chance = 50
			
			# Victory
			change_variable = {
				name = arena_wins
				add = 1
			}
			add_prestige = 50
			add_gold = 25
			
			# Check for rank up
			if = {
				limit = {
					var:arena_wins >= 10
					has_trait = arena_rank_f
				}
				remove_trait = arena_rank_f
				add_trait = arena_rank_e
			}
			if = {
				limit = {
					var:arena_wins >= 25
					has_trait = arena_rank_e
				}
				remove_trait = arena_rank_e
				add_trait = arena_rank_d
			}
		}
		
		# Defeat
		random = {
			chance = 50
			
			change_variable = {
				name = arena_losses
				add = 1
			}
			add_prestige = -25
			
			# Injury chance
			random = {
				chance = 20
				add_trait = wounded
			}
		}
	}
	
	ai_check_interval = 60
	
	ai_potential = {
		is_adult = yes
		is_imprisoned = no
		NOT = { has_trait = wounded }
	}
	
	ai_will_do = {
		base = 10
		
		modifier = {
			factor = 2
			has_trait = brave
		}
		
		modifier = {
			factor = 0.5
			has_trait = craven
		}
	}
}
