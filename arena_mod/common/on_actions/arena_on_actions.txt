# On-actions for arena system
# Note: Duel outcomes are now handled directly in arena_combat.001 event

# Yearly pulse for AI arena participation
yearly_global_pulse = {
	on_actions = {
		arena_ai_participation_check
	}
}

# AI considers entering arena
arena_ai_participation_check = {
	trigger = {
		# Only for valid participants
		arena_valid_participant = yes
		
		# Must be in county with arena
		exists = location
		location = {
			county_has_arena = yes
		}
		
		# Not avoiding arena
		NOT = { should_avoid_arena = yes }
	}
	
	effect = {
		# 10% chance to participate
		random = {
			chance = 10
			
			# Modify by traits
			modifier = {
				factor = 2
				has_trait = brave
			}
			
			modifier = {
				factor = 0.5
				has_trait = craven
			}
			
			modifier = {
				factor = 1.5
				prowess >= 15
			}
			
			# Initialize if needed
			if = {
				limit = { NOT = { has_variable = arena_wins } }
				arena_initialize_participant = yes
			}
			
			# Enter arena
			trigger_event = arena_entry.001
		}
	}
}
