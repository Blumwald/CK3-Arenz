# Arena Combat Events

namespace = arena_combat

# Arena Combat Start - Create opponent and begin duel
arena_combat.001 = {
	type = character_event
	title = arena_combat.001.t
	desc = arena_combat.001.desc
	theme = martial
	
	left_portrait = root
	
	immediate = {
		# Create arena opponent
		create_character = {
			location = root.capital_province
			dynasty = none
			age = { 25 35 }
			gender_female_chance = 50
			random_traits = yes
			trait = education_martial_2
			save_scope_as = arena_opponent
		}
		
		# Start duel with opponent
		start_duel = {
			target = scope:arena_opponent
			skill = prowess
			value = high_duel_value
		}
		
		# Set flag to track this is an arena duel
		set_character_flag = {
			flag = in_arena_duel
			days = 1
		}
	}
	
	option = {
		name = arena_combat.001.a
		# Duel will proceed automatically
	}
}
