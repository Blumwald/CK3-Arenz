namespace = arena_combat

# Match creation and opponent generation
arena_combat.001 = {
	type = character_event
	title = arena_combat.001.t
	desc = arena_combat.001.desc
	theme = martial
	
	left_portrait = {
		character = root
		animation = personality_bold
	}
	
	right_portrait = {
		character = scope:opponent
		animation = personality_bold
	}
	
	immediate = {
		# Generate opponent
		arena_generate_opponent = yes
		
		# Apply class bonuses
		arena_apply_class_bonuses = yes
		
		# Set in-match flag
		add_trait = arena_in_match
		
		# Calculate prowess difference for injury calculation
		set_variable = {
			name = arena_prowess_difference
			value = {
				value = prowess
				subtract = scope:opponent.prowess
			}
		}
		
		# Save opponent to a more persistent scope
		scope:opponent = {
			save_scope_as = arena_opponent
		}
	}
	
	# Begin the duel
	option = {
		name = arena_combat.001.a
		
		# Start vanilla duel with proper scope
		duel = {
			skill = prowess
			target = scope:arena_opponent
			
			# Success (player wins)
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3
					min = -49
				}
				
				desc = arena_combat.001.a.success
				
				# Trigger victory event
				root = {
					trigger_event = arena_resolution.001
				}
				
				# Remove opponent
				scope:arena_opponent = {
					death = { death_reason = death_vanished }
				}
			}
			
			# Failure (player loses)
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3
					min = -49
				}
				
				desc = arena_combat.001.a.failure
				
				# Trigger defeat event
				root = {
					trigger_event = arena_resolution.002
				}
				
				# Remove opponent
				scope:arena_opponent = {
					death = { death_reason = death_vanished }
				}
			}
		}
	}
	
	# Forfeit (lose prestige but no injury risk)
	option = {
		name = arena_combat.001.b
		
		# Lose prestige
		add_prestige = -100
		
		# Gain stress if ambitious/proud
		if = {
			limit = {
				OR = {
					has_trait = ambitious
					has_trait = proud
				}
			}
			add_stress = 20
		}
		
		# Clean up
		arena_cleanup_match = yes
		
		# Kill opponent NPC
		scope:opponent = {
			death = { death_reason = death_vanished }
		}
	}
}
