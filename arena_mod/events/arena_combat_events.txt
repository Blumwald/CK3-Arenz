# Arena Combat Events

namespace = arena_combat

# Arena Combat - Create opponent and initiate duel
arena_combat.001 = {
	type = character_event
	title = arena_combat.001.t
	desc = arena_combat.001.desc
	theme = martial
	
	left_portrait = root
	
	immediate = {
		# Create arena opponent
		create_character = {
			location = root.capital_province
			dynasty = none
			age = { 25 35 }
			gender_female_chance = 50
			random_traits = yes
			trait = education_martial_2
			save_scope_as = arena_opponent
		}
	}
	
	# Fight option - starts the duel
	option = {
		name = arena_combat.001.a
		
		scope:arena_opponent = {
			save_scope_as = arena_opponent
		}
		
		duel = {
			skill = prowess
			target = scope:arena_opponent
			
			# Victory
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				desc = DUEL_PROWESS_DIFF
				
				scope:actor = {
					# Update stats
					change_variable = {
						name = arena_wins
						add = 1
					}
					add_prestige = 50
					add_gold = 25
					
					# Check for rank up to E
					if = {
						limit = {
							var:arena_wins >= 10
							has_trait = arena_rank_f
						}
						remove_trait = arena_rank_f
						add_trait = arena_rank_e
					}
					
					# Check for rank up to D
					if = {
						limit = {
							var:arena_wins >= 25
							has_trait = arena_rank_e
						}
						remove_trait = arena_rank_e
						add_trait = arena_rank_d
					}
				}
				
				# Kill opponent
				scope:arena_opponent = {
					death = {
						death_reason = death_accident
					}
				}
			}
			
			# Defeat
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				desc = DUEL_PROWESS_DIFF
				
				scope:actor = {
					# Update stats
					change_variable = {
						name = arena_losses
						add = 1
					}
					add_prestige = -25
					
					# Injury chance
					random = {
						chance = 30
						add_trait = wounded
					}
				}
				
				# Kill opponent
				scope:arena_opponent = {
					death = {
						death_reason = death_accident
					}
				}
			}
		}
	}
}
