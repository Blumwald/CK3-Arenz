namespace = arena_entry

# Arena entry - Class selection
arena_entry.001 = {
	type = character_event
	title = arena_entry.001.t
	desc = arena_entry.001.desc
	theme = martial
	
	left_portrait = {
		character = root
		animation = personality_bold
	}
	
	immediate = {
		# Check if character already has a class
		# If not, they need to choose one
	}
	
	# Choose Warrior class
	option = {
		name = arena_entry.001.a
		
		trigger = {
			NOT = { has_arena_class = yes }
		}
		
		arena_set_class_warrior = yes
		
		# Proceed to bracket selection
		trigger_event = arena_entry.002
	}
	
	# Choose Mage class
	option = {
		name = arena_entry.001.b
		
		trigger = {
			NOT = { has_arena_class = yes }
		}
		
		arena_set_class_mage = yes
		
		# Proceed to bracket selection
		trigger_event = arena_entry.002
	}
	
	# Choose Rogue class
	option = {
		name = arena_entry.001.c
		
		trigger = {
			NOT = { has_arena_class = yes }
		}
		
		arena_set_class_rogue = yes
		
		# Proceed to bracket selection
		trigger_event = arena_entry.002
	}
	
	# Keep current class (if already chosen)
	option = {
		name = arena_entry.001.d
		
		trigger = {
			has_arena_class = yes
		}
		
		# Proceed to bracket selection
		trigger_event = arena_entry.002
	}
	
	# Cancel
	option = {
		name = arena_entry.001.e
		
		# Do nothing
	}
}

# Bracket selection

	}
