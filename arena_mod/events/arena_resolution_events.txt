namespace = arena_resolution

# Victory event
arena_resolution.001 = {
	type = character_event
	title = arena_resolution.001.t
	desc = arena_resolution.001.desc
	theme = martial
	
	left_portrait = {
		character = root
		animation = personality_bold
	}
	
	right_portrait = {
		character = scope:opponent
		animation = pain
	}
	
	immediate = {
		# Handle victory rewards
		arena_handle_victory = yes
		
		# Update fame modifiers
		arena_update_fame_modifiers = yes
		
		# Clean up match
		arena_cleanup_match = yes
		
		# Kill opponent NPC
		if = {
			limit = { exists = scope:opponent }
			scope:opponent = {
				death = { death_reason = death_vanished }
			}
		}
	}
	
	option = {
		name = arena_resolution.001.a
		
		# Celebrate victory
		add_prestige = 50
	}
	
	option = {
		name = arena_resolution.001.b
		
		trigger = {
			OR = {
				has_trait = humble
				has_trait = calm
			}
		}
		
		# Show humility
		add_piety = 50
	}
}

# Rank up event
arena_resolution.100 = {
	type = character_event
	title = arena_resolution.100.t
	desc = arena_resolution.100.desc
	theme = martial
	
	left_portrait = {
		character = root
		animation = personality_bold
	}
	
	immediate = {
		# Nothing needed
	}
	
	option = {
		name = arena_resolution.100.a
		
		# Gain prestige for rank up
		add_prestige = {
			value = 100
			multiply = var:arena_rank
		}
		
		# Gain fame
		change_variable = {
			name = arena_fame
			add = {
				value = 50
				multiply = var:arena_rank
			}
		}
		
		# Update fame modifiers
		arena_update_fame_modifiers = yes
	}
}

# Remove victor trait after 30 days
arena_resolution.200 = {
	type = character_event
	hidden = yes
	
	immediate = {
		if = {
			limit = { has_trait = arena_victor }
			remove_trait = arena_victor
		}
	}
}
