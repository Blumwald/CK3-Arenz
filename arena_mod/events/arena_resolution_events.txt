# Arena Resolution Events

namespace = arena_resolution

# Arena Victory
arena_resolution.001 = {
	type = character_event
	title = arena_resolution.001.t
	desc = arena_resolution.001.desc
	theme = martial
	
	left_portrait = root
	
	trigger = {
		has_character_flag = in_arena_duel
	}
	
	immediate = {
		remove_character_flag = in_arena_duel
		
		# Update stats
		change_variable = {
			name = arena_wins
			add = 1
		}
		add_prestige = 50
		add_gold = 25
		
		# Check for rank up to E
		if = {
			limit = {
				var:arena_wins >= 10
				has_trait = arena_rank_f
			}
			remove_trait = arena_rank_f
			add_trait = arena_rank_e
		}
		
		# Check for rank up to D
		if = {
			limit = {
				var:arena_wins >= 25
				has_trait = arena_rank_e
			}
			remove_trait = arena_rank_e
			add_trait = arena_rank_d
		}
	}
	
	option = {
		name = arena_resolution.001.a
	}
}

# Arena Defeat
arena_resolution.002 = {
	type = character_event
	title = arena_resolution.002.t
	desc = arena_resolution.002.desc
	theme = martial
	
	left_portrait = root
	
	trigger = {
		has_character_flag = in_arena_duel
	}
	
	immediate = {
		remove_character_flag = in_arena_duel
		
		# Update stats
		change_variable = {
			name = arena_losses
			add = 1
		}
		add_prestige = -25
		
		# Injury chance
		random = {
			chance = 30
			add_trait = wounded
		}
	}
	
	option = {
		name = arena_resolution.002.a
	}
}
